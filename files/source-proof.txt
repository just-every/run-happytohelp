1: #!/bin/sh
2: # Install the rer command (@runeditrun/rer) for this user:
3: #
4: #   curl -fsSL https://runeditrun.com/install.sh | sh
5: #   curl -fsSL https://runeditrun.com/install.sh | sh -s -- dev owner/repo
6: #
7: # It reads the version, tarball and sha512 integrity from the npm registry
8: # (npm_config_registry, default https://registry.npmjs.org; RER_VERSION=X pins an
9: # exact version, otherwise latest), verifies the download, refuses archives with
10: # absolute paths, .. entries, links or devices (and anything tar could read
11: # differently from this script's own check), and installs through the
12: # package's own `rer cli install`: the same layout as
13: # `npx -y @runeditrun/rer cli install`. A failure leaves any previous install
14: # selected and working. It never edits shell profiles: when ~/.local/bin is not
15: # on PATH, the absolute launcher works now, and a run that supplied a command ends
16: # with the one PATH line to run and keep.
17: #
18: # Needs curl and tar. If Node 22 or later is absent, installs the pinned official
19: # Node release in rer's own data directory. macOS/Linux x64/arm64; Windows: WSL2.
20: 
21: # Everything runs from main, so a partly downloaded script does nothing.
